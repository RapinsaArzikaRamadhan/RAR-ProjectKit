package middleware

import (
	"net/http"
)

func CORS(Jawaban http.ResponseWriter, Permintaan *http.Request) bool {
	Jawaban.Header().Set("Access-Control-Allow-Origin", "http://127.0.0.1:5500")
	Jawaban.Header().Set("Access-Control-Allow-Headers", "Content-type")
	Jawaban.Header().Set("Access-Control-Allow-Methods", "POST, DELETE, PUT, GET, OPTIONS")

	if Permintaan.Method == http.MethodOptions {
		return false
	}

	return true
}
