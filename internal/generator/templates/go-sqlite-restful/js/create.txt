
const tombolkirim = document.querySelector("#kirim")

tombolkirim.addEventListener("click", async function () {
    const data = {}

    const response = await fetch("http://127.0.0.1:8080/pengguna", {
        method: "POST",
        headers: { "Content-Type": "Application/json" },
        body: JSON.stringify(data)
    })
    
    if (response.ok) {
        location.reload()
    }
})
