func Pengguna(Jawaban http.ResponseWriter, Permintaan *http.Request) {

	if !middleware.CORS(Jawaban, Permintaan) {
		return
	}

	switch Permintaan.Method {
	case http.MethodPost:
		create(Jawaban,Permintaan)
	case http.MethodGet:
		read(Jawaban,Permintaan)
	case http.MethodPut:
		update(Jawaban,Permintaan)
	case http.MethodDelete:
		delete(Jawaban,Permintaan)
	}
}
