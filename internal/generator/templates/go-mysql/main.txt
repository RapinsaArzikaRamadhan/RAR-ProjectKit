package main

import (
	"database/sql"
	"fmt"
	"net/http"
	"strconv"
	_ "github.com/go-sql-driver/mysql"
)

func koneksidb() (*sql.DB, error) {
	// jangan lupa di ubah sesuai dengan db nye
	dsn := "LatihanMysql:12345@tcp(127.0.0.1:3306)/Latihan1"
	db, err := sql.Open("mysql", dsn)
	
	if err != nil {
		return nil, err
	}

	if err := db.Ping(); err != nil {
		return nil, err
	}	

	return db, nil
}

func perizinanhttp(Jawaban http.ResponseWriter, Permintaan *http.Request) bool {
	Jawaban.Header().Set("Access-Control-Allow-Origin", "http://127.0.0.1:5500")
	Jawaban.Header().Set("Access-Control-Allow-Headers", "HX-Current-Url, HX-Target, HX-Request")

	if Permintaan.Method == http.MethodOptions {
		Jawaban.WriteHeader(http.StatusOK)
		return false
	}

	return true
}

func create(Jawaban http.ResponseWriter, Permintaan *http.Request) {

	if !perizinanhttp(Jawaban, Permintaan) {
		return
	}

	if Permintaan.Method == http.MethodOptions {
		return
	}

	db, err := koneksidb()
	if err != nil {
		fmt.Println(err)
		return
	}
	defer db.Close()

	name := Permintaan.FormValue("name")
	category := Permintaan.FormValue("category")

	qty, err := strconv.Atoi(Permintaan.FormValue("qty"))
	if err != nil {
		fmt.Println(err)
		return
	}

	_, err = db.Exec("insert into items (name, category, qty) value (?,?,?)", name, category, qty)
	
	if err != nil {
    fmt.Println(err)
    return
	}

	fmt.Fprintf(Jawaban, "done kang!")
}

func read(Jawaban http.ResponseWriter, Permintaan *http.Request) {
	if !perizinanhttp(Jawaban, Permintaan) {
		return
	}

	if Permintaan.Method == http.MethodOptions {
		return
	}

	db, err := koneksidb()
	if err != nil {
		fmt.Println(err)
		return
	}
	defer db.Close()

	rows, err := db.Query("select * from items")
	if err != nil {
		fmt.Println(err)
		return
	}
	defer rows.Close()

	for rows.Next() {
		var id, qty int
		var name, category string

		err := rows.Scan(&id,&name,&category,&qty)
		if err != nil {
			fmt.Println(err)
			return
		}

		fmt.Fprintf(Jawaban,
		`
		<tr>
    		<th scope="row">%d</th>
    		<td>%s</td>
    		<td>%s</td>
    		<td>%d</td>
		</tr>
		`,id, name, category, qty)
	}

	if err := rows.Err(); err != nil {
		fmt.Println(err)
		return
	}

}

func update(Jawaban http.ResponseWriter, Permintaan *http.Request) {
	if !perizinanhttp(Jawaban, Permintaan) {
		return
	}
	
	if Permintaan.Method == http.MethodOptions {
		return
	}

	db,err := koneksidb()
	if err != nil {
		fmt.Println(err)
		return
	}
	defer db.Close()

	id,err := strconv.Atoi(Permintaan.FormValue("id"))
	if err != nil{
		fmt.Println(err)
		return
	}
	name := Permintaan.FormValue("name")
	category := Permintaan.FormValue("category")
	qty,err := strconv.Atoi(Permintaan.FormValue("qty"))
	if err != nil{
		fmt.Println(err)
		return
	}

	_,err = db.Exec("update items set name = ?, category = ?, qty = ? where id = ?", name, category, qty, id)
	if err != nil{
		fmt.Println(err)
		return
	}

	fmt.Fprintf(Jawaban, "done kang!")

}

func delete(Jawaban http.ResponseWriter, Permintaan *http.Request) {
	if !perizinanhttp(Jawaban, Permintaan) {
		return
	}

	if Permintaan.Method == http.MethodOptions {
		return
	}

	db, err := koneksidb()
	if err != nil {
		fmt.Println(err)
		return
	}
	defer db.Close()

	id, err := strconv.Atoi(Permintaan.FormValue("id"))
	if err != nil {
		fmt.Println(err)
		return
	}
	_, err = db.Exec("delete from items where id = ?", id)

	if err != nil {
    fmt.Println(err)
    return
	}

	fmt.Fprintf(Jawaban, "berhasil!")
}

func main() {
	http.HandleFunc("/create", create)
	http.HandleFunc("/read", read)
	http.HandleFunc("/delete", delete)
	http.HandleFunc("/update", update)
	http.ListenAndServe(":8080", nil)
}
